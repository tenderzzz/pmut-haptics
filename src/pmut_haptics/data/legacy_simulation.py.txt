def clamped_plate_fundamental(D, rho, h, a):
    """Clamped 圆板基频. \u03bb\u00b2=10.2158 (Leissa, NASA SP-160)"""

# 2.2: 2D 声场云图
def rayleigh_2d(xg, zg, k, a, rho0, c0, u0=1e-6, N=50):
    r_vals = np.linspace(0, a, N)
    dr = a / N
    p = np.zeros_like(xg, dtype=complex)
    for i, ri in enumerate(r_vals):
        dS = np.pi * (dr/2)**2 if i == 0 else 2*np.pi*ri*dr
        R = np.sqrt(xg**2 + zg**2 + ri**2 - 2*xg*ri)
        R = np.maximum(R, 1e-12)
        p += (u0*dS/(2*np.pi)) * 1j*k*rho0*c0 * np.exp(-1j*k*R) / R
    return p

def far_field_dir(theta, k, a):
    x = k * a * np.sin(theta)
    res = np.ones_like(x)
    mask = np.abs(x) > 1e-10
    res[mask] = 2 * special.jv(1, x[mask]) / x[mask]
    return np.abs(res)

# --- 3.1: 聚焦延迟计算 ---

def focusing_delays(array_x, F, c):
    """
    对每个阵元计算聚焦延时.
    几何关系: 第 n 个阵元到焦点 (0, F) 的距离为 sqrt(x_n^2 + F^2)
    需要补偿的额外传播时间为 sqrt(x_n^2 + F^2)/c - F/c
    """
    return np.sqrt(array_x**2 + F**2) / c - F / c

delays = focusing_delays(array_x, F_focal, c0)

for n, t_delay in enumerate(delays):
    print(f"    阵元 {n}: x = {array_x[n]*1e3:7.1f} mm,  delay = {t_delay*1e6:7.2f} us")

# --- 3.2: 阵列聚焦声场计算 ---

def array_piston_field(xg, zg, k, a, array_x, delays, u0=1e-6, N_div=20):
    """
    计算 1xN 线阵在聚焦相位下的合成声场.
    每个阵元视为圆形活塞, 用 Rayleigh-Sommerfeld 积分离散求和.
    delays: 每个阵元的延时(秒), 通过 e^(j*omega*delay) 转化为相位.
    """
    p_total = np.zeros_like(xg, dtype=complex)
    omega_k = k * c0

    for n, x_n in enumerate(array_x):
        # 该阵元相对于场点的位置
        x_rel = xg - x_n
        R = np.sqrt(x_rel**2 + zg**2)

        # 圆形活塞自身的方向性 (远场近似, 直接用于近场简化模型)
        # 对每个阵元内的小面元积分, 这里用解析近似:
        # 单个活塞在距离 R 处的压力 ~ (j*rho*c*k/(2*pi)) * u0 * area * D(theta) * exp(-jkR)/R
        theta_n = np.arctan2(x_rel, zg)
        sin_theta = np.sin(theta_n)
        x_bessel = k * a * sin_theta
        directivity = np.ones_like(x_bessel)
        mask = np.abs(x_bessel) > 1e-10
        directivity[mask] = 2 * special.jv(1, x_bessel[mask]) / x_bessel[mask]

        area = np.pi * a**2
        R_safe = np.maximum(R, 1e-12)

        # 聚焦相位: 先到达焦点者先发射 (补偿), 即乘以 e^{+j*omega*delay}
        phase = 1j * k * c0 * delays[n]
        p_total += (1j * k * rho0 * c0 * u0 * area / (2 * np.pi)
                    * directivity * np.exp(-1j * k * R_safe) / R_safe
                    * np.exp(1j * phase))

    return p_total

# --- 4.1: 合成孔径回波模拟 ---

def simulate_synthetic_aperture(array_x, scatterers, k, c, t_span, fs,
                                pulse_width=3, u0=1e-6):
    """
    合成孔径模式: 每个阵元依次发射, 所有阵元同时接收.
    返回 rf_data: shape (N_tx, N_rx, N_t)
    """
    N_elem = len(array_x)
    N_scat = len(scatterers)
    N_t = len(t_span)
    dt = 1.0 / fs

    rf = np.zeros((N_elem, N_elem, N_t))

    for tx in range(N_elem):
        for scat_idx, (xs, zs) in enumerate(scatterers):
            # 发射路径: 阵元 tx -> 散射体
            dist_tx = np.sqrt((array_x[tx] - xs)**2 + zs**2)
            # 接收路径: 散射体 -> 阵元 rx
            dist_rx = np.sqrt((array_x - xs)**2 + zs**2)
            total_dist = dist_tx + dist_rx
            travel_time = total_dist / c

            # 简化脉冲: 3 个周期的正弦波包络
            # 每个接收阵元独立计算
            for rx in range(N_elem):
                # 信号在 travel_time 处到达
                t_center = travel_time[rx]
                # 使用高斯包络调制正弦脉冲
                sigma = pulse_width / (2 * np.pi * k * c / (2*np.pi))
                # 简化: 用固定时间宽度
                t_rel = t_span - t_center
                pulse = np.exp(-(t_rel / (pulse_width * 2e-5))**2) * np.sin(2*np.pi*k*c/(2*np.pi) * t_rel)
                # 幅度: 1/R^2 (球面衰减)
                amplitude = u0 * scatterer_strength[scat_idx] / (dist_tx * dist_rx[rx] + 1e-12)
                rf[tx, rx, :] += amplitude * pulse

    return rf

# 时间轴参数
fs = 1e6          # 采样率 1 MHz (对 23 kHz 信号足够)
t_max = (2 * np.sqrt(60e-3**2 + 40e-3**2)) / c_img + 50e-6
t_span = np.arange(0, t_max, 1/fs)

print(f"\n  时间采样: fs = {fs/1e3:.0f} kHz, 时长 = {t_max*1e3:.2f} ms, 点数 = {len(t_span)}")
print(f"  合成孔径仿真中...")

rf_data = simulate_synthetic_aperture(
    array_x, scatterers, k_img, c_img, t_span, fs,
    pulse_width=3, u0=1e-6
)

print(f"  回波数据维度: {rf_data.shape}")

# --- 4.2: 延时叠加波束合成 ---

def delay_and_sum_beamform(rf_data, array_x, t_span, c, x_grid, z_grid):
    """
    合成孔径延时叠加成像:
    对每个成像点 (xi, zi), 计算所有 (tx, rx) 组合的往返时间,
    取对应时刻的回波幅度求和.
    """
    N_elem = len(array_x)
    image = np.zeros((len(x_grid), len(z_grid)))

    for i, xi in enumerate(x_grid):
        for j, zi in enumerate(z_grid):
            total = 0.0
            for tx in range(N_elem):
                dist_tx = np.sqrt((array_x[tx] - xi)**2 + zi**2)
                for rx in range(N_elem):
                    dist_rx = np.sqrt((array_x[rx] - xi)**2 + zi**2)
                    travel_time = (dist_tx + dist_rx) / c
                    idx_t = int(travel_time * fs)
                    if 0 <= idx_t < len(t_span):
                        total += rf_data[tx, rx, idx_t]
            image[i, j] = np.abs(total)

    return image

# 成像网格
x_img_grid = np.linspace(x_img_min, x_img_max, 80)
z_img_grid = np.linspace(z_img_min, z_img_max, 80)
